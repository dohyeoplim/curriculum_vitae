\newcommand{\metadata}[1]{{\color{black!60}#1}}

\newcommand{\entrylayout}[4]{
  #1\hfill \metadata{#2}\\[1pt]
  {\small #3\hfill \metadata{#4}}\par\vspace{4pt}
}

\newcommand{\entry}[4]{
  \entrylayout{\textbf{#1}}{#2}{#3}{#4}
}

\newcommand{\publicationentry}[3]{
  \begingroup
  \hyphenpenalty=10000
  #1\\[1pt]
  {\small #2\enspace\metadata{#3}}\par\vspace{4pt}
  \endgroup
}

\newcommand{\dateditem}[2]{
  \item #1\hfill \metadata{#2}\par
}

\newcommand{\itemdetail}[1]{
  \vspace{-1pt}{\footnotesize\metadata{#1}\par}
}

\newcommand{\skillline}[2]{
  \item{\small{#1:} #2\par}
}

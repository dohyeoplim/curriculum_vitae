\newcommand{\cvheader}[2]{%

  \noindent

\newenvironment{cvsection}[1]{%
  \noindent
  \begin{minipage}[t]{0.17\linewidth}%
    \vspace{0pt}%
    \raggedright
    \setstretch{0.80}%
    {\footnotesize\scshape\color{accent}\lsstyle #1}%
  \end{minipage}%
  \hfill
  \begin{minipage}[t]{0.81\linewidth}%
    \vspace{0pt}%
}{%
  \end{minipage}%
  \par
  \cvsectiongap
}

% #1 title, #2 meta, #3 subtitle (optional), #4 sub-meta
\newcommand{\cventry}[4]{%
  \entrymark{\small #1}\hfill{\scriptsize\color{muted}#2}\par
  \ifblank{#3}{}{%
    \entrylinegap
    {\footnotesize\itshape #3}\hfill{\scriptsize\color{muted}#4}\par
  }%
}

% #1 title, #2 authors and venue
\newcommand{\publicationentry}[2]{%
  \entrymark{\footnotesize #1}\par
  \entrylinegap
  {\scriptsize\color{muted}#2}\par
  \cventrygap
}

% #1 name, #2 date, #3 description, #4 url (optional)
\newcommand{\projectitem}[4]{%
  \entrymark{\footnotesize #1}%
  \ifblank{#4}{}{\,\href{#4}{\scriptsize\color{accent}[link]}}%
  \hfill{\scriptsize\color{muted}#2}\par
  \entrylinegap
  {\footnotesize\color{muted}#3}\par
  \cventrygap
}

% #1 label, #2 value

  \end{minipage}\par
  \vspace{2pt}
}

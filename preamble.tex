\definecolor{accent}{HTML}{7A3B2F}
\definecolor{muted}{HTML}{3B3733}
\definecolor{detailgray}{cmyk}{0,0,0,0.90}
\definecolor{rulegray}{HTML}{938B82}

\hypersetup{hidelinks}
\pagestyle{empty}

\setstretch{1.07}
\setlength{\parindent}{0pt}
\setlength{\parskip}{0pt}

\newcommand{\entrylinegap}{\vspace{0.5pt}}
\newcommand{\cventrygap}{\par\addvspace{4pt}}
\newcommand{\cvipgap}{\cventrygap}
\newcommand{\cvgroupgap}{\par\addvspace{6pt}}
\newcommand{\cvprojectgap}{\cventrygap}
\newcommand{\cvsectiongap}{\vspace{16pt}}
\newcommand{\cvsubsectiongap}{\vspace{1pt}}

% Shared text styles and inline metadata.
\newcommand{\cvemphasis}[1]{\textsbf{#1}}
\newcommand{\cvmetastyle}{\footnotesize\color{muted}}
\newcommand{\cvmeta}[1]{{\cvmetastyle #1}}
\newcommand{\cvstatus}[1]{%
  \ifblank{#1}{}{\enspace{\cvmetastyle\itshape #1}}%
}
\newcommand{\cvlabel}[1]{{\footnotesize\textcolor{rulegray}{[#1]}}}

% Numbered entries wrap back to the label's left edge.
\newcommand{\cvnumberedpara}[2]{%
  {\small\raggedright
    \noindent\cvlabel{#1}\hspace{3pt}#2\par
  }%
}

% Optional URL, then #2 title, #3 date.
\newcommand{\cvtitleline}[3][]{%
  {\small\raggedright
    #2\ifblank{#1}{}{\,\href{#1}{\footnotesize\color{accent}[link]}}%
    \hfill\cvmeta{#3}\par
  }%
}

% #1 name, #2 contacts
\newcommand{\cvsep}{\;\textcolor{rulegray}{\textperiodcentered}\;}
\newcommand{\cvheader}[2]{%
  {\centering
    {\Huge #1\par}%
    \vspace{6pt}%
    {\small\color{muted}#2\par}%
    \vspace{9pt}%
    \nointerlineskip
    \noindent\textcolor{rulegray}{\rule{\linewidth}{0.5pt}}\par
  }%
  \vspace{18pt}%
  \nointerlineskip
}

\newenvironment{cvsection}[1]{%
  \noindent
  \begin{minipage}[t]{0.17\linewidth}%
    \vspace{0pt}%
    \raggedright
    \setstretch{0.80}%
    {\small\scshape\color{accent}\lsstyle #1\par}%
  \end{minipage}%
  \hfill
  \begin{minipage}[t]{0.81\linewidth}%
    \vspace{0pt}%
}{%
  \par\unskip
  \end{minipage}%
  \par
  \cvsectiongap
  \nointerlineskip
}

\newenvironment{cvsubsection}[1]{%
  {\footnotesize\sffamily\bfseries\color{muted}#1\par}%
  \cvsubsectiongap
}{%
  \par
}

% Compact bullet details use a narrower indent than numbered entries.
\newlength{\cvdetailindent}
\setlength{\cvdetailindent}{10pt}
\newcommand{\cvbulletpara}[1]{%
  {\small\color{detailgray}\raggedright
    \leftskip=\cvdetailindent
    \noindent\llap{\makebox[\dimexpr\cvdetailindent-2pt\relax][c]{\footnotesize\textcolor{rulegray}{$\bullet$}}\hspace{2pt}}#1\par
  }%
}

\newcommand{\cvbulletdetail}[1]{%
  \entrylinegap
  \cvbulletpara{#1}%
}

% #1 title, #2 date, #3 subtitle
\newcommand{\cventry}[3]{%
  \cvtitleline{#1}{#2}%
  \cvbulletdetail{\itshape #3}%
}

\newcounter{publication}

% Optional contribution, then #2 title, #3 authors, #4 year, #5 status.
\newcommand{\publicationentry}[5][]{%
  \stepcounter{publication}%
  \cvnumberedpara{\arabic{publication}}{%
    #3. ``#2.'' #4.\cvstatus{#5}}%
  \ifblank{#1}{}{\cvbulletdetail{#1}}%
  \cventrygap
}

% Auxiliary information stays aligned with its title, without a bullet.
\newcommand{\cvaux}[1]{%
  \entrylinegap
  {\cvmetastyle\raggedright #1\par}%
}

% #1 identifier, #2 title, #3 status and date
\newcommand{\ipentry}[3]{%
  \cvnumberedpara{#1}{#2.\cvstatus{#3}}%
  \cvipgap
}

% Optional detail rows, then #2 title and #3 participation period.
\newcommand{\researchproject}[3][]{%
  \cvtitleline{#2}{#3}%
  \ifblank{#1}{}{#1}%
  \cvprojectgap
}

% Optional URL, then #2 name, #3 date, #4 description.
\newcommand{\projectitem}[4][]{%
  \cvtitleline[#1]{#2}{#3}%
  \cvdetail{#4}%
  \cventrygap
}

% #1 text, #2 date; starred form drops the trailing gap
\makeatletter
\newcommand{\dated@norm}[2]{\cvtitleline{#1}{#2}\cventrygap}
\newcommand{\dated@tight}[2]{\cvtitleline{#1}{#2}}
\newcommand{\dateditem}{\@ifstar\dated@tight\dated@norm}
\makeatother

\newcommand{\cvdetail}[1]{%
  \entrylinegap
  {\raggedright\small\color{muted} #1\par}
}
